import {useEffect,useRef,useState} from 'react';
import {activityLog,type ActivityEntry,type ActivityKind,type ActivityPage} from './game-api';
import {activityDateRange} from './activity-core';import {todayKST} from './core';
const labels:Record<ActivityKind,string>={all:'전체',pomodoro:'뽀모도로',quiz:'오답 복습',purchase:'상점 구매'};
const time=(date:string)=>new Intl.DateTimeFormat('ko-KR',{timeZone:'Asia/Seoul',hour:'2-digit',minute:'2-digit',hour12:false}).format(new Date(date));
const day=(value:string)=>new Intl.DateTimeFormat('ko-KR',{timeZone:'Asia/Seoul',year:'numeric',month:'long',day:'numeric',weekday:'short'}).format(new Date(value+'T12:00:00+09:00'));
export function ActivityLog(){
 const [kind,setKind]=useState<ActivityKind>('all');const [search,setSearch]=useState('');const [debounced,setDebounced]=useState('');const [from,setFrom]=useState('');const [to,setTo]=useState('');const [refresh,setRefresh]=useState(0);
 const [entries,setEntries]=useState<ActivityEntry[]>([]);const [summary,setSummary]=useState<ActivityPage['summary']|null>(null);const [hasMore,setHasMore]=useState(false);const [loading,setLoading]=useState(true);const [more,setMore]=useState(false);const [error,setError]=useState('');const generation=useRef(0);
 useEffect(()=>{const timer=setTimeout(()=>setDebounced(search.trim()),300);return()=>clearTimeout(timer);},[search]);
 useEffect(()=>{const updated=()=>setRefresh(value=>value+1);window.addEventListener('study-reward',updated);return()=>window.removeEventListener('study-reward',updated);},[]);
 useEffect(()=>{let live=true;generation.current++;setLoading(true);setMore(false);setError('');setEntries([]);setSummary(null);setHasMore(false);try{const range=activityDateRange(from,to);void activityLog({kind,search:debounced,...range}).then(data=>{if(live){setEntries(data.entries);setSummary(data.summary);setHasMore(data.hasMore);}}).catch(error=>{if(live)setError(error.message);}).finally(()=>{if(live)setLoading(false);});}catch(error){setError(error instanceof Error?error.message:'날짜를 확인해주세요.');setLoading(false);}return()=>{live=false;};},[kind,debounced,from,to,refresh]);
 async function loadMore(){if(more||!entries.length)return;const current=generation.current;setMore(true);setError('');try{const data=await activityLog({kind,search:debounced,...activityDateRange(from,to)},entries.at(-1));if(current===generation.current){setEntries(old=>{const seen=new Set(old.map(row=>row.kind+row.id));return [...old,...data.entries.filter(row=>!seen.has(row.kind+row.id))];});setSummary(data.summary);setHasMore(data.hasMore);}}catch(error){if(current===generation.current)setError(error instanceof Error?error.message:'기록을 불러오지 못했습니다.');}finally{if(current===generation.current)setMore(false);}}
 const grouped=new Map<string,ActivityEntry[]>();for(const entry of entries){const date=todayKST(new Date(entry.occurred_at));grouped.set(date,[...(grouped.get(date)||[]),entry]);}
 return <><header className="game-heading"><div><span className="eyebrow">MY LITTLE STUDY JOURNAL</span><h1>활동 기록</h1><p>언제 무엇을 공부했고, 어떤 작은 물건을 데려왔는지.</p></div><button className="btn secondary" disabled={loading||more} onClick={()=>setRefresh(value=>value+1)}>새로고침</button></header>
 <section className="activity-summary" aria-label="활동 합계"><div><span>받은 코인</span><strong>{summary?'+'+Number(summary.earned).toLocaleString():'—'}</strong></div><div><span>쓴 코인</span><strong>{summary?'−'+Number(summary.spent).toLocaleString():'—'}</strong></div><div><span>현재 보유 코인</span><strong>{summary?Number(summary.balance).toLocaleString():'—'}</strong></div></section>
 <section className="game-section activity-filters"><div className="activity-tabs" role="group" aria-label="기록 종류">{(['all','pomodoro','quiz','purchase'] as ActivityKind[]).map(value=><button className={kind===value?'selected':''} aria-pressed={kind===value} key={value} onClick={()=>setKind(value)}>{labels[value]}</button>)}</div><div className="activity-query"><label className="sr-only" htmlFor="activity-search">활동 검색</label><input id="activity-search" value={search} onChange={event=>setSearch(event.target.value)} placeholder="과목, 문제, 목표, 구매한 물건 검색"/><label>시작일<input type="date" value={from} onChange={event=>setFrom(event.target.value)} aria-label="기록 시작일"/></label><label>종료일<input type="date" value={to} onChange={event=>setTo(event.target.value)} aria-label="기록 종료일"/></label>{(search||from||to||kind!=='all')&&<button className="text-btn" onClick={()=>{setSearch('');setDebounced('');setFrom('');setTo('');setKind('all');}}>초기화</button>}</div><p className="activity-count">{loading?'기록을 불러오는 중…':`총 ${Number(summary?.count||0).toLocaleString()}건`}{!!summary?.unknownPurchases&&` · 이전 구매 중 금액이 남아 있지 않은 기록 ${summary.unknownPurchases}건`}</p></section>
 {error&&<p className="error" role="alert">{error}</p>}
 {!loading&&!error&&!entries.length&&<div className="settings-empty activity-empty"><span>✧</span><h2>아직 표시할 기록이 없어요.</h2><p>공부 보상을 받거나 상점에서 물건을 사면 이곳에 차곡차곡 남습니다.</p></div>}
 <div className="activity-timeline">{[...grouped].map(([date,rows])=><section className="activity-day" key={date}><h2>{day(date)}</h2><div className="activity-entries">{rows.map(entry=><article className={`activity-entry activity-${entry.kind}`} key={entry.kind+entry.id}><div className="activity-symbol" aria-hidden="true">{entry.kind==='pomodoro'?'◷':entry.kind==='quiz'?'✓':'✦'}</div><div className="activity-body"><div className="activity-row-top"><span className="activity-kind">{labels[entry.kind]}</span><time dateTime={entry.occurred_at}>{time(entry.occurred_at)}</time></div><h3>{entry.title}</h3>{entry.kind==='pomodoro'&&<><p>{entry.details.cycle?`${entry.details.cycle}회차 · `:''}{entry.minutes}분 집중 완료</p>{entry.details.goal&&<blockquote>{entry.details.goal}</blockquote>}</>}{entry.kind==='quiz'&&<><p>{[entry.details.page&&`책 p.${entry.details.page}`,entry.details.questionNumber&&`${entry.details.questionNumber}번`,typeof entry.details.correct==='boolean'&&(entry.details.correct?'정답':'오답')].filter(Boolean).join(' · ')}</p>{entry.details.question&&<blockquote>{entry.details.question}</blockquote>}</>}{entry.kind==='purchase'&&<p>{entry.details.legacy?'이전 구매 기록':'상점에서 구매 · 보관함에 저장'}</p>}{entry.kind!=='purchase'&&<small className="activity-breakdown">기본 {entry.base} + 랜덤 {entry.random_bonus}{entry.correct_bonus?` + 정답 ${entry.correct_bonus}`:''}</small>}</div><strong className={`activity-amount ${entry.kind==='purchase'?'spent':'earned'}`}>{entry.amount===null?'금액 기록 없음':`${entry.amount>0?'+':'−'}${Math.abs(entry.amount).toLocaleString()}코인`}</strong></article>)}</div></section>)}</div>
 {hasMore&&<div className="activity-more"><button className="btn secondary" disabled={more} onClick={()=>void loadMore()}>{more?'불러오는 중…':'이전 기록 더 보기'}</button></div>}
 </>;
}
